#Список переменных
#папка исходников
SRC= src/
#папка с хедерами
INC= include/
#папка с объектными файлами
OBJ= obj/
#папка с исполняемым файлом
BLD= build/
#название исп. файла
EXE= $(BLD)lab5_crc
#компилятор
CC= gcc
#флаги компилятора
CFLAGS= -c -Wall -I$(INC) -MMD
#-c         - только компиляция
#-Wall      - все предупреждения
#-g         - точки останова
#-I$(INC)   - указание для компилятора пути до хедер файлов
#-MMD       - компилятор будет сохранять файлы зависимостей в папке с объектными
#================================================================

#Главная цель
#patsubst - функция заменит шаблон $(SRC)/%.c на шаблон $(OBJ)%.o
#для всех строк из списка $(wildcard $(SRC)*.c)
#wildcard - функция найдет в указанной директории все файлы по шаблону *.c
#и вернет список всех найденных через пробел
#% - символ означает любую строку
#вместо этого символа make подставит по очереди каждую строку из списка
#который выдаст wildcard, например, функция выдала src/main.c src/fun1.c src/fun2.c
#вместо % выберется соответственно main fun1 fun2
all: $(patsubst $(SRC)%.c, $(OBJ)%.o, $(wildcard $(SRC)*.c))
	$(CC) $^ -o $(EXE)

#=====Шаблонное правило=====
#Когда make будет работать с главной целью понадобится либо пересобрать файл либо
#создать его. Будет найдена ближайшее подходящее правило или самое конкретное.
#В данном случае будет выбрано это правило и вместо % будет подставлено имя целевого файла
#---------------------------
#$^ - автоматическая переменная, которая заменяется первым слева файлом зависимости
#S@ - автоматическая переменная, которая заменяется именем цели
$(OBJ)%.o: $(SRC)%.c
	$(CC) $(CFLAGS) $< -o $@
#===========================

#=====Директива подключения файлов зависимостей=====
#
include $(wildcard $(OBJ)*.d)
#===================================================

clean:
	rm -f $(OBJ)*.o $(OBJ)*.d

#Специальная цель, которая говорит make что это название цели, а не файла
#Нужно на случай если вдруг появится файл с именем all или clean
.PHONY: all clean
